let map, autocomplete, selectedPlace, userLocation; let markers = []; function initMap() { // Default to Youngstown if GPS fails const defaultCenter = { lat: 41.1030, lng: -80.6514 }; map = new google.maps.Map(document.getElementById("map"), { zoom: 12, center: defaultCenter, disableDefaultUI: true }); // 1. GET USER GPS & CENTER MAP if (navigator.geolocation) { navigator.geolocation.getCurrentPosition( (position) => { userLocation = { lat: position.coords.latitude, lng: position.coords.longitude }; map.setCenter(userLocation); loadBars("Monday"); // Trigger load once we have location }, () => { console.log("GPS denied, using default."); loadBars("Monday"); } ); } // 2. AUTOCOMPLETE FOR ADDING BARS const input = document.getElementById("bar-search"); autocomplete = new google.maps.places.Autocomplete(input); autocomplete.addListener("place_changed", () => { const place = autocomplete.getPlace(); if (!place.geometry) return; selectedPlace = { name: place.name, address: place.formatted_address, lat: place.geometry.location.lat(), lng: place.geometry.location.lng() }; }); } // 3. NAVIGATION FEATURE function startNavigation(lat, lng) { // Opens native Google Maps app for turn-by-turn window.open(`https://www.google.com/maps/dir/?api=1&destination=${lat},${lng}`, '_blank'); } async function loadBars(day) { const res = await fetch(`/api/specials?day=${day}`); const data = await res.json(); const list = document.getElementById("specialsList"); list.innerHTML = ""; markers.forEach(m => m.setMap(null)); markers = []; data.forEach(bar => { // 4. RADIUS FILTER: 45 MILES const distance = calculateDistance(userLocation, {lat: bar.lat, lng: bar.lng}); if (userLocation && distance > 45) return; const card = document.createElement("div"); card.className = "card"; // Click the card to navigate card.onclick = () => startNavigation(bar.lat, bar.lng); card.innerHTML = ` ${bar.name} ${bar.special}
${bar.address}
Tap to Navigate (${distance.toFixed(1)} mi)
`; list.appendChild(card); const marker = new google.maps.Marker({ position: { lat: bar.lat, lng: bar.lng }, map: map, title: bar.name }); markers.push(marker); }); } // Distance calculation (Haversine Formula) function calculateDistance(p1, p2) { if (!p1 || !p2) return 0; const R = 3958.8; // Miles const dLat = (p2.lat - p1.lat) * Math.PI / 180; const dLon = (p2.lng - p1.lng) * Math.PI / 180; const a = Math.sin(dLat/2) * Math.sin(dLat/2) + Math.cos(p1.lat * Math.PI / 180) * Math.cos(p2.lat * Math.PI / 180) * Math.sin(dLon/2) * Math.sin(dLon/2); const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)); return R * c; }
